cmake_minimum_required(VERSION 3.16)
project(TernaryTreeLab VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_GUI "Build the Qt Widgets interface" ON)
option(BUILD_GUI_TESTS "Build Qt Test interface checks" ON)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UBSan (GCC/Clang)" OFF)

add_library(tree_core INTERFACE)
target_include_directories(tree_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_library(tree_workspace src/Workspace.cpp)
target_link_libraries(tree_workspace PUBLIC tree_core)
if(MSVC)
    target_compile_options(tree_workspace PUBLIC /utf-8 /W4)
else()
    target_compile_options(tree_workspace PUBLIC -Wall -Wextra -Wpedantic)
endif()
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(tree_workspace PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(tree_workspace PUBLIC -fsanitize=address,undefined)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(tree_tests tests/test_main.cpp tests/test_tree.cpp tests/test_workspace.cpp)
    target_link_libraries(tree_tests PRIVATE tree_workspace)
    add_test(NAME tree_tests COMMAND tree_tests)
endif()
add_executable(tree_benchmarks benchmarks/benchmarks.cpp)
target_link_libraries(tree_benchmarks PRIVATE tree_core)
if(MSVC)
    target_compile_options(tree_benchmarks PRIVATE /utf-8)
endif()

if(BUILD_GUI)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
    set(CMAKE_AUTOMOC ON)
    add_library(tree_ui src/MainWindow.cpp src/MainWindow.h)
    target_link_libraries(tree_ui PUBLIC tree_workspace Qt${QT_VERSION_MAJOR}::Widgets)
    target_include_directories(tree_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
    if(QT_VERSION_MAJOR EQUAL 6)
        qt6_add_executable(TernaryTreeQt MANUAL_FINALIZATION src/main.cpp)
    else()
        add_executable(TernaryTreeQt src/main.cpp)
    endif()
    target_link_libraries(TernaryTreeQt PRIVATE tree_ui)
    if(WIN32)
        set_target_properties(TernaryTreeQt PROPERTIES WIN32_EXECUTABLE TRUE)
    endif()
    if(QT_VERSION_MAJOR EQUAL 6)
        qt6_finalize_target(TernaryTreeQt)
    endif()
    if(BUILD_TESTING AND BUILD_GUI_TESTS)
        find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
        add_executable(tree_gui_tests tests/test_gui.cpp)
        target_link_libraries(tree_gui_tests PRIVATE tree_ui Qt${QT_VERSION_MAJOR}::Test)
        add_test(NAME tree_gui_tests COMMAND tree_gui_tests -platform offscreen)
    endif()
endif()
